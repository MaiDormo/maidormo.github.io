\projecthead{dist-patoh --- Distributed GPU Hypergraph Partitioning}{M.S. thesis, 2026--27}
\begin{itemize}[nosep, after=\strut, leftmargin=1.5em, itemsep=1pt, label=\color{accent}\textbullet]
    \item Built a distributed multilevel hypergraph partitioner in \textbf{\texttt{C++20}/CUDA/MPI}: distributed coarsening, FM/Jet refinement, and recursive-bisection k-way partitioning; \textbf{126/126} cluster runs passed correctness and balance at P=4--8, k=2--64 (2$\times$ A30, L40S).

\end{itemize}
