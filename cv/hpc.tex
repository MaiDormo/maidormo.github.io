\section{Research \& Projects}

\projecthead{dist-patoh: Distributed GPU Hypergraph Partitioning}{M.S. Thesis (In Progress), 2026--2027}
\begin{itemize}[nosep, after=\strut, leftmargin=1.5em, itemsep=1pt, label=\color{accent}\textbullet]
    \item Developing a distributed multi-GPU hypergraph partitioner in \textbf{C++20, CUDA, and MPI} to split large, irregular sparse graphs across GPUs while balancing compute load and minimizing cross-node communication.
    \item Implementing a multilevel pipeline (parallel coarsening, recursive k-way partitioning, and local refinement) to handle the load imbalance and block-scheduling bottlenecks of standard sparse kernels.
    \item Validated GPU coarsening and refinement across 4--8 MPI ranks on A30/L40S nodes, with correctness checks passing and measured speedups over CPU baselines; extending the pipeline to larger multi-GPU runs.
\end{itemize}

\projectdivider

\projectheadcode{Hybrid Adaptive SpMV, CUDA}{https://github.com/MaiDormo/cuda-SpMV}{GPU Computing, 2025}
\begin{itemize}[nosep, after=\strut, leftmargin=1.5em, itemsep=1pt, label=\color{accent}\textbullet]
    \item Wrote and benchmarked CSR SpMV kernels on an NVIDIA A30, combining thread-per-row and warp-per-row execution with read-only cache loads (\texttt{\_\_ldg}) and warp-shuffle reductions.
    \item Best measured 649 GB/s (about 70\% of peak HBM bandwidth); matched or beat cuSPARSE on several irregular matrices and diagnosed block-overscheduling on very large sparse graphs.
\end{itemize}

\projectdivider

\projectheadcode{Parallel Minimum Spanning Tree, MPI + OpenMP}{https://github.com/MaiDormo/parallel-mst}{HPC, 2025}
\begin{itemize}[nosep, after=\strut, leftmargin=1.5em, itemsep=1pt, label=\color{accent}\textbullet]
    \item Implemented a hybrid MPI + OpenMP Boruvka MST in C for dense graphs using a flat, cache-aligned symmetric adjacency matrix, memory-mapped I/O (mmap), and bulk collectives.
    \item Evaluated 2--32 MPI ranks on a 64-core node (best measured 12.9$\times$ on 40k vertices); the flat layout avoids pointer chasing, and mmap plus one bulk broadcast replace the baseline's per-row input path.
\end{itemize}
